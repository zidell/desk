cmake_minimum_required(VERSION 3.20)
project(DeskGuard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(KWin REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Qml Widgets Quick)
add_library(deskguard MODULE guard.cpp)
set_target_properties(deskguard PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_link_options(deskguard PRIVATE "LINKER:-z,defs")
target_link_libraries(deskguard PRIVATE KWin::kwin Qt6::DBus Qt6::Qml xkbcommon)
